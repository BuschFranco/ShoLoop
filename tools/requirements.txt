# Dependencies for the asset generators in tools/.
#
#   pip install -r tools/requirements.txt
#
# gen_sfx.py, gen_sprites.py and build_assets.py are stdlib-only by design, so a fresh clone can
# regenerate every sound and every silhouette with nothing installed. Only two generators need a
# library, and each needs exactly one.
#
# Both are pinned to a minimum rather than an exact version: they use long-stable basics, and pinning
# exactly would break anyone whose distro ships something newer.

# prep_character_sprite.py, to turn a photograph into a character portrait -- the one job that
# genuinely needs an imaging library.
Pillow>=10.0

# gen_font.py, to assemble Assets/Fonts/PixelFont.ttf. Pure Python, no compiler, no system font
# tooling. Needed only to *rebuild* the font; the .ttf is committed, so playing and building the game
# needs nothing installed.
fonttools>=4.40
